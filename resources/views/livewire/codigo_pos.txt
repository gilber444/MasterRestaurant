    public function ScanCode($barcode, $cant = 1)
    {
        //dd($barcode);
        $p = Precios::find($barcode);
        $product = Productos::join('precios as p', 'productos.id', '=', 'p.producto')
            ->join('inventarios as i', 'i.producto', 'productos.id')
            ->where('p.id',  $p->id )
            //->where('p.cantidad', 1)
            ->where('i.sucursal', session('sucursal'))
            ->select('productos.id', 'productos.nombreProducto', 'i.existencia', 'p.codebar', 'i.id as inventario', 'p.pvventa', 'p.cantidad', 'i.sucursal', 'p.presentacion', 'p.cantidad as descargar', 'p.medida')
            ->first();


        if ($product == null) {
            $this->emit('scan-notfound', 'El producto no encontrado');
        } else {
            $user_id = Auth::user()->id;
            $price = $p->pvventa;
            $des = Descuentos::where('producto', $product->id)
                ->where('inicio', '<=', Carbon::today()->endOfDay())
                ->where('fin', '>=', Carbon::today()->startOfDay())
                ->latest('created_at')
                ->first();

            if ($des) {
                $descuento = ($price * $des->descuento) / 100;
                $sub = $price - $descuento;
            } else {
                $descuento = 0;
                $sub = $price;
            }

            $existingCartItem = tmpVentas::where('producto', $product->id)
            ->where('user', $user_id)
            ->where('sucursal', session('sucursal'))
            ->first();

            if ($existingCartItem)
            {
                if ($existingCartItem->descargar + $cant > $product->existencia) {
                    $this->emit('scan-notfound', 'Stock insuficiente');
                }
                else
                {
                    $existingCartItem->quantity = $existingCartItem->quantity + $cant;
                    $existingCartItem->total = ($existingCartItem->quantity) * $existingCartItem->price;
                    $existingCartItem->descargar = ($existingCartItem->quantity + $cant) * $existingCartItem->limit;
                    $existingCartItem->save();
                    $this->CantiUpdate($product->id, $existingCartItem->id);
                }
            }
            else
            {
                //dd($cant * $product->descargar + $cant);
                if ($cant > $product->existencia) {
                    $this->emit('scan-notfound', 'Stock insuficiente');
                }
                else
                {
                    $tmp =tmpVentas::create([
                        'producto' => $product->id,
                        'name' => $product->nombreProducto,
                        'price' => $sub,
                        'quantity' => $cant,
                        'sucursal' => $product->sucursal,
                        'codebar' => $product->codebar,
                        'descuento' => $descuento,
                        'total' => $sub * $cant,
                        'medida' => $product->presentacion,
                        'limit' => $product->descargar,
                        'descargar' => $cant * $product->descargar,
                        'uni' => $product->medida,
                        'user' => $user_id,
                        'caja' => session('caja'),
                        'empresa' => session('empresa'),
                        'esenario' => 1
                    ]);
                    //$this->CantiUpdate($product->id, $tmp->id);
                }
            }
            $this->Carrito();
        }
    }
    ////////////para la escala de prodcutos
    public function CantiUpdate($id, $tmp)
    {

        $user_id = Auth::user()->id;
        $items = tmpVentas::find($tmp);

        $bus = Precios::where('producto', $id)
            ->where('codebar', $items->codebar)
            ->where('cantidad', '<', $items->quantity)
            ->orderBy('cantidad', 'desc')
            ->first();

        // Si no se encuentra en la escala más baja, hacer una segunda consulta
        if (!$bus) {
            $bus = Precios::where('producto', $id)
                ->where('codebar', $items->codebar)
                ->where('cantidad', '>=', $items->quantity)
                ->orderBy('cantidad', 'asc')
                ->first();
        }

        $items->price = $bus->pventasiva;
        //$items->quantity = $this->can[$tmp];
        $items->total = $items->quantity * $bus->pventasiva;
        $items->limit = $bus->cantidad;
        $items->descargar = $bus->cantidad * $items->quantity;
        $items->save();

        $this->Carrito();
    }

    public function updateCanti($id)
    {
        $user_id = Auth::user()->id;

        $tmp = tmpVentas::find($id);
        // Obtener el producto junto con su inventario
        $product = Productos::join('inventarios as i', 'i.producto', 'productos.id')
        ->where('sucursal', session('sucursal'))
        ->find($tmp->producto);
        // Cantidad solicitada
        $cantidades = $this->can[$id];

        // Verificar si hay suficiente stock
        if ($cantidades > $product->existencia) {
            $this->emit('scan-notfound', 'Stock insuficiente');
        } else {
            // Buscar descuento si aplica
            $des = Descuentos::where('producto', $tmp->producto)
                ->where('inicio', '<=', now())
                ->where('fin', '>=', now())
                ->latest('created_at')
                ->first();

            if ($des) {
                $descuento = ($tmp->price * $des->descuento) / 100;
                $sub = $tmp->price - $descuento;
            } else {
                $descuento = 0;
                $sub = $tmp->price;
            }

            $tmp->descargar = $tmp->limit * $cantidades;
            $tmp->quantity = $cantidades;
            $tmp->total = $cantidades * $sub;
            $tmp->save();
            $this->UpdateEscala($tmp->producto, $tmp->id);
            $this->Carrito();
        }
    }

    //////////actualiza la medida para los precios
    public function updateUni($id)
    {
        // Imprime el valor actual del select para asegurarte de que se está actualizando
        if (!isset($this->uni[$id])) {
            return;
        }

        // Muestra el valor seleccionado del select para este $id
        $selectedValue = $this->uni[$id];
        //dd($selectedValue); // Verifica el valor seleccionado

        // Busca el dato en la tabla temporal tmpVentas
        $datos = tmpVentas::find($id);

        if (!$datos) {
            return;
        }
        // Busca el precio en la tabla de precios basado en la medida seleccionada
        $precios = Precios::find($selectedValue);

        if (!$precios) {
            return;
        }

        // Resto del proceso de actualización
        $descargar = $precios->cantidad * $datos->quantity;

        $exis = Inventarios::where('producto', $datos->producto)
                            ->where('sucursal', session('sucursal'))
                            ->first();

        if ($exis && $exis->existencia >= $descargar) {
            $medida = Medidas::find($precios->medida);

            $datos->price = $precios->pvventa;
            $datos->limit = $precios->cantidad;
            $datos->descargar = $descargar;
            $datos->total = $precios->pvventa * $datos->quantity;
            $datos->medida = $medida->unidad;
            $datos->uni = $precios->medida;
            $datos->save();

            // Actualiza el carrito después de guardar los cambios
            $this->carrito();
        } else {
            $this->emit('scan-notfound', 'Stock insuficiente');
        }
    }


    //////escala de precios al actualizar la cantidad
    public function UpdateEscala($id, $tmp)
    {

        $user_id = Auth::user()->id;
        $items = tmpVentas::find($tmp);

        $bus = Precios::where('producto', $id)
            ->where('codebar', $items->codebar)
            ->where('cantidad', '<=', $items->quantity)
            ->where('medida', $items->uni)
            ->orderBy('cantidad', 'desc')
            ->first();

        // Si no se encuentra en la escala más baja, hacer una segunda consulta
        if (!$bus) {
            $bus = Precios::where('producto', $id)
                ->where('codebar', $items->codebar)
                ->where('cantidad', '>=', $items->quantity)
                ->where('medida', $items->uni)
                ->orderBy('cantidad', 'asc')
                ->first();
        }

        $items->price = $bus->pvventa;
        //$items->quantity = $this->can[$tmp];
        $items->total = $items->quantity * $bus->pvventa;
        //$items->limit = $items->cantidad;
        //$items->descargar = $items->limit * $items->quantity;
        $items->save();

        $this->Carrito();
    }

    public function ScanCodeById($barcode)
    {
        $cant = 1;
        $product = Productos::join('precios as p', 'productos.id', '=', 'p.producto')
        ->join('inventarios as i', 'i.producto', 'productos.id')
        ->where('p.codebar',  $barcode)
        ->where('p.cantidad', 1)
        ->where('i.sucursal', session('sucursal'))
        ->select('productos.id', 'productos.nombreProducto', 'i.existencia', 'p.codebar', 'i.id as inventario', 'p.pvventa', 'p.cantidad', 'i.sucursal', 'p.presentacion', 'p.cantidad as descargar', 'p.medida')
        ->first();

        if ($product == null) {
            $this->emit('scan-notfound', 'El producto no encontrado');
        } else {
            $user_id = Auth::user()->id;
            $price = $product->pvventa;
            $des = Descuentos::where('producto', $product->id)
                ->where('inicio', '<=', Carbon::today()->endOfDay())
                ->where('fin', '>=', Carbon::today()->startOfDay())
                ->latest('created_at')
                ->first();

            if ($des) {
                $descuento = ($price * $des->descuento) / 100;
                $sub = $price - $descuento;
            } else {
                $descuento = 0;
                $sub = $price;
            }

            $existingCartItem = tmpVentas::where('producto', $product->id)
            ->where('user', $user_id)
            ->where('sucursal', session('sucursal'))
            ->first();

            if ($existingCartItem)
            {
                if ($existingCartItem->descargar + $cant >= $product->existencia) {
                    $this->emit('scan-notfound', 'Stock insuficiente');
                }
                else
                {
                    $existingCartItem->quantity = $existingCartItem->quantity + $cant;
                    $existingCartItem->total = ($existingCartItem->quantity + $cant) * $existingCartItem->price;
                    $existingCartItem->descargar = ($existingCartItem->quantity + $cant) * $existingCartItem->limit;
                    $existingCartItem->save();
                    $this->CantiUpdate($product->id, $existingCartItem->id);
                }
            }
            else
            {
                if ($cant > $product->existencia) {
                    $this->emit('scan-notfound', 'Stock insuficiente');
                }
                else
                {
                    $tmp =tmpVentas::create([
                        'producto' => $product->id,
                        'name' => $product->nombreProducto,
                        'price' => $sub,
                        'quantity' => $cant,
                        'sucursal' => $product->sucursal,
                        'codebar' => $product->codebar,
                        'descuento' => $descuento,
                        'total' => $sub * $cant,
                        'medida' => $product->presentacion,
                        'limit' => $product->descargar,
                        'descargar' => $cant * $product->descargar,
                        'uni' => $product->medida,
                        'user' => $user_id,
                        'caja' => session('caja'),
                        'empresa' => session('empresa'),
                        'esenario' => 1
                    ]);
                    $this->CantiUpdate($product->id, $tmp->id);
                }
            }

            $this->Carrito();
        }
    }
